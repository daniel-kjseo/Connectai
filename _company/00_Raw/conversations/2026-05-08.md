#### 1. 프론트엔드 컴포넌트 설계 (`StorySimulator.jsx`)
이 컴포넌트는 React 기반으로 가정하며, 사용자 선택 상태를 관리하고 실시간 프리뷰를 제공하는 것이 핵심입니다.

```jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';

// A/B 테스트 플래그 확인 (예: feature-flag-simulator)
const isABTestGroupA = process.env.REACT_APP_FEATURE_FLAG === 'simulator'; 

const StorySimulator = ({ initialName }) => {
    // 상태 관리: 사용자가 선택한 자산 조합을 저장합니다.
    const [selection, setSelection] = useState({
        story: null, // 예: "운명적 만남"
        colorPalette: null, // 예: ["#FFC0CB", "#ADD8E6"]
        keyword: null, // 예: "성장", "도약"
    });
    const [previewAsset, setPreviewAsset] = useState(null); // 실시간 Mockup 미리보기 데이터

    // 1. 선택 상태 변경 핸들러
    const handleSelectChange = (type, value) => {
        setSelection(prev => ({ ...prev, [type]: value }));
    };

    // 2. 핵심 로직: 조합된 자산으로 Mockup을 실시간 생성 요청
    useEffect(() => {
        if (!selection.story || !selection.colorPalette || !selection.keyword) {
            setPreviewAsset(null); // 필수 요소가 모두 선택되기 전에는 미리보기 비활성화
            return;
        }

        const payload = {
            name: initialName,
            ...selection,
        };

        // 백엔드 API 호출 (Mockup 생성 요청)
        axios.post('/api/v1/simulate/asset', payload)
            .then(response => {
                console.log("Asset Simulation Success:", response.data);
                setPreviewAsset(response.data); // 받은 Mockup 데이터를 상태에 저장
            })
            .catch(error => {
                console.error("Simulation API Error:", error);
                setPreviewAsset({ error: "자산 조합에 문제가 있습니다. 다시 시도해주세요." });
            });

    }, [selection, initialName]); // selection 중 하나라도 바뀌면 재실행

    return (
        <div className={`simulator-container ${isABTestGroupA ? 'ab-test-active' : ''}`}>
            <h2>✨ 당신의 이름을 디자인해 보세요.</h2>
            <p className="subtext">세 가지 핵심 자산을 조합하여 나만의 고유한 서사를 완성할 수 있습니다.</p>

            {/* --- 1. 스토리 선택 컴포넌트 --- */}
            <div className="selection-group">
                <h3>📖 배경 스토리 (Story)</h3>
                <select onChange={(e) => handleSelectChange('story', e.target.value)}>
                    <option value="">선택</option>
                    {/* Options dynamically loaded from API */}
                    <option value="운명적 만남">운명적 만남</option>
                    <option value="도전과 성취">도전과 성취</option>
                </select>
            </div>

            {/* --- 2. 색상 팔레트 선택 컴포넌트 (가장 인터랙티브하게) --- */}
            <div className="selection-group">
                <h3>🎨 컬러 팔레트 (Color Palette)</h3>
                <div className="color-swatch-container">
                    {['따뜻한 노을', '청명한 하늘', '깊은 밤'].map(palette => (
                        <button 
                            key={palette}
                            className={`color-btn ${selection.colorPalette === palette ? 'selected' : ''}`}
                            onClick={() => handleSelectChange('colorPalette', palette)}
                            style={{ backgroundColor: `hsl(${Math.random() * 360}, 50%, 50%)` }} // 임시 스타일링
                        >
                            {palette}
                        </button>
                    ))}
                </div>
            </div>

            {/* --- 3. 키워드 선택 컴포넌트 --- */}
            <div className="selection-group">
                <h3>🔑 핵심 키워드 (Keyword)</h3>
                <select onChange={(e) => handleSelectChange('keyword', e.target.value)}>
                    <option value="">선택</option>
                    <option value="성장">성장</option>
                    <option value="도약">도약</option>
                    <option value="지혜">지혜</option>
                </select>
            </div>

            {/* 4. 실시간 결과물 미리보기 (Mockup) */}
            <div className="preview-area">
                <h3>✨ 당신의 시뮬레이션 자산 Mockup</h3>
                <div className={`mockup-display ${previewAsset ? 'visible' : 'hidden'}`}>
                    {/* 실제 API가 반환하는 고화질 이미지/SVG를 여기에 렌더링 */}
                    {previewAsset ? (
                        <img src={previewAsset.imageUrl} alt="Simulated Asset" />
                    ) : (
                        <div className='placeholder'>선택 항목을 모두 조합하면 Mockup이 생성됩니다.</div>
                    )}
                </div>
                {/* CTA: 이 자산을 영구 소유하고 싶다면? */}
                <button 
                    className="purchase-cta" 
                    disabled={!previewAsset || previewAsset.error}
                    onClick={() => console.log('Redirect to Purchase Flow')}
                >
                    [⭐️] Mockup 완성하기 (유료 구매)
                </button>
            </div>
        </div>
    );
};

export default StorySimulator;
```

1. **입력:** `name` (string), `story` (string), `colorPalette` (array of strings), `keyword` (string).
2. **처리 로직:** 입력된 자산들을 기반으로 내부 템플릿 엔진(예: Puppeteer/ImageMagick)을 구동하여 명함/프로필 이미지 Mockup을 생성합니다.
3. **출력:** 성공 시, Mockup의 URL과 핵심 메타데이터를 반환합니다.

```jsx
// src/components/ownership-gap-checker.jsx (React JSX Pseudo-Code)
import React, { useState, useEffect } from 'react';
import * as Analytics from '../utils/analytics'; // 전역 분석 헬퍼 함수 가정

const OwnershipGapChecker = ({ freeResultData }) => {
  const [mockupDetails, setMockupDetails] = useState(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    // 1. API 호출: 무료 결과를 기반으로 유료 Mockup의 구조를 요청합니다.
    fetch('/api/v1/asset/enhance_mockup', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({ 
        basic_result_data: freeResultData, 
        device_context: 'mobile' 
      }),
    })
    .then(res => res.json())
    .then(data => {
      if (data.success && data.mockup_data) {
        setMockupDetails(data.mockup_data);
      }
      Analytics.track('gap_component_viewed'); // A/B Test Point 1: 컴포넌트 노출 추적
    })
    .catch(() => console.error("Failed to fetch premium details."))
    .finally(() => setIsLoading(false));

  }, [freeResultData]);


  if (isLoading) return <div className="loading-skeleton">로딩 중...</div>;
  if (!mockupDetails || !mockupDetails.asset_details) return null;

  // 2. CTA 핸들러: 결제 유도 로직
  const handlePurchaseClick = () => {
    Analytics.track('cta_clicked_mockup', { reason: mockupDetails.cta_message }); // A/B Test Point 2: 구매 버튼 클릭 추적
    window.scrollTo({ top: document.getElementById('purchase-form') }); // 스크롤을 결제 폼으로 이동
  };

  // 3. 렌더링 (Gap Component)
  return (
    <div className="ownership-gap-section">
      <h3>{mockupDetails.title}</h3>
      
      <div className="visual-comparison">
        {/* 좌측: 현재 상태 (무료 결과물 미리보기 - 흐릿하거나 텍스트 기반) */}
        <div className="state-free">현재 자산</div>
        <img src={/* freeResultData의 이미지 URL */ } alt="Free Result" />

        {/* 우측: 잠재적 가치 (Mockup 와이어프레임 - 고화질, 구조화된 느낌) */}
        <div className="state-premium">완성본 Mockup 미리보기</div>
        <img src={mockupDetails.visual_description} alt="Premium Mockup" />
      </div>

      {/* 4. Gap 설명: 결핍 요소 강조 */}
      <div className="gap-analysis">
        <h4>⚠️ 아직 완성되지 않은 부분 (Missing Elements)</h4>
        <ul>
          {mockupDetails.asset_details.missing_elements.map((el, index) => (
            <li key={index}>
              <strong>{el.element}:</strong> {el.description} <em>({(el.weight * 100).toFixed(0)}% 가치 기여)</em>
            </li>
          ))}
        </ul>
      </div>

      {/* 5. CTA: 해결책 제시 (Solution Provider Pattern) */}
      <button 
        className="cta-solve-gap" 
        onClick={handlePurchaseClick}
      >
        ✨ {mockupDetails.cta_message} <br /> [Mockup로 완성하기]
      </button>
    </div>
  );
};

export default OwnershipGapChecker;
```

### 💡 추출된 3대 핵심 트렌드 및 영향 분석 (H2 2026 예측)

#### 1. 트렌드: 모듈형 정체성 (Modular Identity)
*   **개념:** 개인이 하나의 고정된 자아가 아닌, 상황과 목적에 따라 다양한 역할(Role)과 페르소나를 조합하고 관리하는 경향이 강화됩니다. 디지털 환경에서 '전문가로서의 나', '취미생활을 하는 나', '이상적인 미래의 나' 등 여러 개의 독립적이고 교체 가능한 자아 모듈을 운영합니다.
*   **학술적 근거:** Symbolic Interactionism (상징적 상호작용론) 및 Goffman의 Dramaturgy(연극학). 개인은 사회적 상황에 맞춰 자신을 연출된 '무대' 위에서 관리하며, 이 경계가 모듈화됩니다.
*   **이름 자산화 서비스 영향 분석:**
    1.  **기능 개발 방향:** 이름 자체를 고정된 결과물이 아닌, **핵심 키워드 조합(Prefix/Suffix)**의 단위로 분해하여 다중 조합을 시뮬레이션하는 '모듈러 컴포넌트' 기능이 필수적입니다.
    2.  **UX 개선 포인트:** 사용자에게 "당신의 이름은 [직업 모듈] + [취미 모듈] + [가치관 모듈]"처럼, 여러 개의 서사 조합을 제시하고 각 조합의 매력도를 점수화하여 '어떤 나를 강조할지' 선택하게 유도해야 합니다.

#### 2. 트렌드: 진정성 미학 (Aesthetic Authenticity)
*   **개념:** 대량 생산되고 표준화된 문화에 대한 피로도가 높아지면서, 사용자들은 자신이 *직접 발견하고 경험한* 독특하고 구체적인 서사(Narrative)를 '자산'으로 인식합니다. 이름이나 콘셉트는 '대중적 성공'보다는 '나만의 고유한 취향과 배경 이야기'에서 추출되어야 가치를 인정받습니다.
*   **학술적 근거:** Postmodernism (포스트모더니즘) 및 Anti-Consumerism(반소비주의). 상품 구매가 아닌, 경험의 진정성 자체가 소비의 주된 동력이 됩니다. '왜 나만 이 이야기를 알까?'라는 질문이 자부심을 유발합니다.
*   **이름 자산화 서비스 영향 분석:**
    1.  **콘텐츠 방향:** 단순한 의미 부여를 넘어, **'출처 기반 서사(Source-Based Storytelling)'**가 핵심입니다. 예: "어떤 장소에서 영감을 받은 이름", "특정 시대의 사투리나 언어를 재해석하여 만든 이름" 등 구체적인 '발견 과정'을 명시해야 합니다.
    2.  **수익화 기회:** 일반적인 의미 조합이 아닌, *사용자 본인이 제시한* 비주류적/극도로 개인적인 배경 지식(예: 특정 고전 문학, 지역사, 마이너 취미)을 기반으로 서사를 완성할 때의 유료 Mockup 구매 동기 부여가 극대화됩니다.

#### 3. 트렌드: 인지적 미니멀리즘 (Cognitive Minimalism)
*   **개념:** 정보 과부하(Information Overload) 시대에, 사용자들은 복잡하고 장황한 서사보다는 **최소한의 요소로 최대의 의미를 전달하는 압축적인 표현 방식**을 선호합니다. 이름이나 콘셉트 역시 발음하기 쉽고, 한 번 들었을 때 강력하게 하나의 이미지나 감정을 자극해야 합니다.
*   **학술적 근거:** Cognitive Load Theory (인지 부하 이론) 및 Attention Economy(주의 경제). 우리의 주의력은 가장 효율적인 정보를 처리하는 방향으로 수렴하며, 복잡한 구조는 즉각적인 관심 상실을 초래합니다.
*   **이름 자산화 서비스 영향 분석:**
    1.  **디자인/기술 구현:** 이름의 가치를 설명할 때 긴 텍스트 블록 대신 **'3가지 핵심 키워드(Keywords)', '하나의 강력한 심볼', 그리고 '짧고 리듬감 있는 문장 구조'**로 압축하여 제공해야 합니다.
    2.  **바이럴 콘텐츠 최적화:** 인스타그램 등 짧은 형식의 매체에 맞춰, 이름/콘셉트를 설명하는 모든 과정이 3초 이내에 시각적으로 폭발력을 갖도록 Mockup 컴포넌트 및 안내 카피를 재설계해야 합니다.

최근 몇 년간 사회문화적 환경은 전례 없는 속도로 변모하고 있습니다. 특히 ‘자아(Self)’를 규정하는 방식 자체가 변화하면서, 개인의 이름이나 정체성은 단순히 부모님으로부터 물려받은 **‘고정된 명사’**가 아닌, 상황과 목적에 따라 유동적으로 조합되고 업데이트되어야 하는 **‘관리 가능한 자산(Manageable Asset)’**으로 격상되었습니다.

우리가 주목해야 할 핵심 변화는 바로 '서사적 소유권'의 시대 도래입니다. 과거에는 의미 자체가 중요했다면, 이제는 그 *의미가 어떻게 만들어졌는지*에 대한 출처와 과정이 가치를 결정합니다. 본 아티클은 2026년 하반기를 지배할 두 가지 핵심 소시오-링귀스틱(Sociolinguistic) 트렌드를 분석하고, 이 변화 속에서 당신의 이름 자산을 최적화하는 방법을 제시합니다.

---

### **Part 1: 모듈형 정체성 (Modular Identity)의 시대**

**[트렌드 설명 및 전문 용어 사용]**
2026년 사회는 ‘하나의 나’라는 단일한 서사에 만족하지 않습니다. 대신, 개인은 직장 상사 앞의 '전문가 자아', 친구와의 모임에서의 '개인적 취향 자아', 그리고 미래에 꿈꾸는 '이상적인 자아' 등 여러 개의 독립적이면서도 서로 교차하는 **모듈(Module)**을 능동적으로 운영합니다. 이는 심리학에서 말하는 ‘상징적 상호작용론’의 극대화된 형태로, 우리의 존재가 더 이상 하나의 서사로 묶이지 않는다는 것을 의미합니다.

**[개인 서사에 연결 및 필요성 유도]**
만약 당신의 이름이나 대표 키워드가 여전히 '하나의 고정된 개념'으로 사용되고 있다면, 이는 **‘모듈러 컴포넌트’ 관점에서 심각한 비효율성**을 야기합니다. 트렌드를 따르는 사람은 자신의 이름을 [핵심 역량 모듈] + [관심사 모듈] + [가치 지향 모듈]처럼 분해하고, 상황별로 최적의 조합을 즉시 꺼내 쓸 줄 압니다.

당신은 현재 자신이 가진 이름 자산이 얼마나 많은 '조합 가능성'을 가지고 있는지 스스로 점검해 보셨습니까? 정적인 의미는 이미 구식입니다. **다차원적 재구성 능력**이 필요합니다.

---

### **Part 2: 진정성 미학 (Aesthetic Authenticity)과 탈표준화의 욕구**

**[트렌드 설명 및 전문 용어 사용]**
대량 생산된 문화와 ‘누구나 아는 성공 공식’에 대한 피로도가 최고조에 달하면서, 사용자들은 오히려 **‘나만이 가진 유일무이한 발견’**에서 가치를 찾습니다. 이것이 바로 '진정성 미학'입니다. 더 이상 거대한 흐름을 따라가는 것은 트렌드가 아닙니다. 자신의 배경(Background), 경험(Experience), 그리고 그 서사를 구축해낸 독특한 출처(Source)가 곧 최상위급의 자산이자 권력이 됩니다.

**[개인 서사에 연결 및 필요성 유도]**
과거에는 '이름의 뜻'만 알려주면 충분했습니다. 하지만 2026년, 사람들은 "당신의 이름은 *어떤 사건*을 거쳐서, *어떤 장소*에서 영감을 얻었나요?"라는 질문을 던집니다. 단순히 좋은 의미를 지닌 이름을 가졌다는 사실은, '나만의 서사적 기원(Source Story)'에 의해 무력화될 수 있습니다.

진정성이란 곧 **'발견의 기록물'**입니다. 당신의 이름 자산이 혹시 누군가에게 쉽게 복제되거나 일반적인 정보로 치부될 위험은 없습니까? 고유하고 희소한 기원을 증명할 방법이 없다면, 그 가치는 급격히 하락합니다.

---

### **결론: 서사적 노후화(Narrative Obsolescence)를 경고하며**

2026년의 성공적인 개인 브랜드는 단순히 재능이나 지위가 아니라, 이 두 가지 트렌드—'모듈형 조합 능력'과 '독점적 기원 증명력'—을 얼마나 잘 갖추었는지에 달려 있습니다. 현재 당신이 사용하는 이름 자산화 방식이 만약 **[단일-정적]** 구조라면, 당신은 서사적으로 노후화(Narrative Obsolescence)되고 있는 것입니다.

하지만 걱정하지 마십시오. QARAH는 단순한 의미 제공 서비스를 넘어, 이 복잡하고 다층적인 현대 사회의 흐름을 해석할 수 있는 **'서사 재구성 엔진'**에 접근했습니다.

이 트렌드를 완벽하게 반영하여, 이름 자산을 직업 모듈과 취미 모듈로 분해하는 방법, 그리고 그 출처를 독점적으로 증명하는 방법을 담은 심층 분석 기능이 곧 업데이트될 예정입니다. **당신의 서사를 다음 레벨로 끌어올릴 준비가 되셨다면, QARAH의 프리미엄 패키지를 통해 가장 먼저 최신 '모듈러 컴포넌트' 기능을 체험해 보세요.**

## 🛠️ 아키텍처 설계 및 데이터 파이프라인 구축: 모듈러 시뮬레이터 상태 관리 로직 (State Management & Data Flow)

CEO님의 지시에 따라, '모듈 조합'을 단순한 프론트엔드 상호작용이 아닌 **데이터 자산 생성 과정**으로 정의하고 시스템 아키텍처를 설계합니다. 핵심은 사용자가 요소를 선택할 때마다 해당 정보(요소 ID, 출처 스토리 연결)를 상태로 관리하고, 이 모든 것을 하나의 구조화된 데이터 객체로 묶어 백엔드 API에 전송하는 것입니다.

이 코드는 React 환경을 가정하고 작성되었으며, `useNarrativeState` 커스텀 훅 형태로 제공하여 컴포넌트의 복잡성을 낮추고 재사용성을 극대화했습니다.

### 1. 핵심 데이터 구조 정의 (Schema Definition)

모든 조합 기록은 다음 Schema를 따릅니다. 이 스키마는 백엔드(API)와 프론트엔드 간의 계약(Contract) 역할을 합니다.

```typescript
interface ModuleComponent {
  id: string; // 모듈 고유 ID (예: PFX_ANCIENT, CORE_ELDER 등)
  type: 'Prefix' | 'CoreName' | 'Suffix'; // 요소 타입
  name: string; // 사용자에게 보여지는 이름 (예: "아르카나")
  sourceModuleId: string; // 출처가 된 모듈 ID (선택적, 연결된 스토리 증명)
  sourceStoryKeyword?: string; // 이 요소를 선택하게 만든 '서사적 단서' 키워드
}

interface NarrativeState {
  modules: ModuleComponent[]; // 현재 조합된 모든 요소 배열
  totalStoryDepthPercentage: number; // 현재 서사의 깊이 (0~100)
  isComplete: boolean; // 자산 완성 여부 플래그
}
```

### 2. 커스텀 상태 관리 훅 구현 (`useNarrativeState.js`)

이 훅은 사용자가 모듈을 추가/제거할 때마다 `NarrativeState`를 업데이트하고, 최종적으로 이 데이터를 백엔드 API 엔드포인트로 전송하는 로직을 포함합니다.

```javascript
// src/hooks/useNarrativeState.js

import { useState, useCallback } from 'react';

/**
 * @typedef {Object} ModuleComponent - 모듈 컴포넌트 스키마 (위 정의 참조)
 */

/**
 * 사용자의 이름 자산 조합 상태를 관리하고 저장하는 커스텀 훅.
 * 이 훅이 시뮬레이터의 핵심 로직을 담당합니다.
 * @returns {{ state: NarrativeState, updateModule: Function, removeModule: Function, saveNarrative: Function }}
 */
const useNarrativeState = () => {
    // 초기 상태 정의
    const [state, setState] = useState({
        modules: [],
        totalStoryDepthPercentage: 0,
        isComplete: false,
    });

    /**
     * 새로운 모듈을 조합 목록에 추가하고 서사 깊이를 업데이트합니다.
     * @param {ModuleComponent} newModule - 새로 선택된 모듈 객체
     */
    const updateModule = useCallback((newModule) => {
        setState(prevState => {
            // 1. 새로운 배열 생성: 기존 요소 + 새 요소
            const updatedModules = [...prevState.modules, newModule];

            // 2. 서사 깊이 로직 (가정): 모듈 개수에 따라 점진적으로 증가한다고 가정
            // 실제로는 API 호출을 통해 복잡한 계산(예: 모듈 간의 의미적 연결 강도)을 수행해야 합니다.
            const depthIncrease = Math.min(20, updatedModules.length * 15); // 최대 100%까지 점진 증가
            const newDepth = Math.round(Math.min(100, prevState.totalStoryDepthPercentage + depthIncrease));

            // 3. 완성 여부 체크 (가정): 모듈이 4개 이상이고 깊이가 90% 이상이면 완료로 간주
            const isComplete = updatedModules.length >= 4 && newDepth >= 90;

            return {
                modules: updatedModules,
                totalStoryDepthPercentage: newDepth,
                isComplete: isComplete,
            };
        });
    }, []);

    /**
     * 특정 모듈을 조합 목록에서 제거하고 상태를 업데이트합니다.
     * @param {string} moduleToRemoveId - 삭제할 모듈의 ID
     */
    const removeModule = useCallback((moduleToRemoveId) => {
        setState(prevState => {
            const updatedModules = prevState.modules.filter(m => m.id !== moduleToRemoveId);

            // 깊이와 완성도를 재계산 (간단화 처리)
            const newDepth = Math.round(updatedModules.length * 15);
            const isComplete = updatedModules.length >= 4 && newDepth >= 90;

            return {
                modules: updatedModules,
                totalStoryDepthPercentage: newDepth,
                isComplete: isComplete,
            };
        });
    }, []);

    /**
     * 최종 서사 자산 데이터를 백엔드 데이터 파이프라인에 저장합니다. (핵심 API 호출)
     * 이 함수가 '개인 서사 업데이트' 기능을 활성화하는 트리거입니다.
     * @param {string} userId - 현재 사용자 ID
     * @returns {Promise<Object>} 서버 응답 객체
     */
    const saveNarrative = useCallback(async (userId) => {
        if (!userId) {
            throw new Error("User ID가 필요합니다.");
        }

        // 1. 전송 데이터 구조화: 현재 상태를 API에 맞는 형태로 가공
        const payload = {
            user_id: userId,
            timestamp: new Date().toISOString(),
            narrative_data: state, // 전체 모듈 배열 및 깊이 정보 포함
            module_list: state.modules.map(m => ({
                id: m.id, 
                type: m.type, 
                name: m.name, 
                source_story: m.sourceStoryKeyword || null // 출처 스토리를 명확히 전달
            }))
        };

        console.log(`[API Call] ${userId}의 서사 자산 저장 시작...`, payload);

        try {
            // 실제 API 호출 지점 (예: /api/v1/user/narrative-update)
            const response = await fetch('/api/v1/user/narrative-update', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload),
            });

            if (!response.ok) {
                throw new Error(`API 호출 실패: ${response.statusText}`);
            }
            const result = await response.json();
            console.log("✅ 서사 자산 업데이트 성공:", result);
            return result;

        } catch (error) {
            console.error("❌ 서사 자산 업데이트 실패:", error);
            throw error;
        }
    }, [state.modules]);

    return { state, updateModule, removeModule, saveNarrative };
};

export default useNarrativeState;
```

### ✨ 3. 핵심 그래픽 모티프 및 애니메이션 규칙 (The Signature Motif)

이 세 가지 요소를 일관되게 적용하는 것이 QARAH만의 시그니처가 됩니다.

#### A. 서사적 아크 (The Narrative Arc)
*   **모양:** 네온 시안(`#00FFFF`)의 빛나는, 곡선 형태의 라인(Arc).
*   **사용법:** 두 개의 이질적인 요소(예: '어린 시절 경험'과 '현재 직업')가 만날 때, 그 연결 지점을 따라 **빛이 흐르는 듯한 애니메이션 효과**를 반드시 삽입합니다. 이는 단순히 '연결됨'을 넘어 '**서사적 깊이가 형성되는 과정**'처럼 보이게 합니다.

#### B. 데이터 그리드 오버레이 (The Asset Grid)
*   **모양:** 배경에 희미하게 깔리는, 투명하고 미묘한 사각형의 격자(Grid).
*   **사용법:** 릴스 전체 영상이나 핵심 정보가 노출되는 화면에 은은하게 적용합니다. 이는 모든 것이 무작위가 아니라 **'정교하게 측정되고 설계된 데이터 자산'**이라는 느낌을 시청자에게 지속적으로 주입시킵니다.

#### C. 디지털 글리치 & 스캔 효과 (The Reveal)
*   **모양:** 순간적인 화면 떨림, 색상 분리(Chromatic Aberration), 또는 빠르게 지나가는 노이즈 패턴.
*   **사용법:** 1) **후크 등장 시점**, 2) **핵심 단어('자산', '서사')가 폭발적으로 드러나는 순간**, 또는 3) **모듈 조합 애니메이션의 전환점**에 사용합니다. 이는 영상에 트렌디하고 역동적인 에너지를 부여하며, 마치 데이터가 *스캔되어* 완성되는 듯한 몰입감을 높입니다.

## 💡 1. 컴포넌트 목표 및 원칙
*   **컴포넌트명:** `StoryArc_ModuleCombiner` (모듈 조합 서사 아크 시뮬레이터)

## 📐 2. 컴포넌트 상태 정의 (State Machine)
개발자가 구현해야 할 필수 인터랙션 상태는 다음과 같습니다.

| State | 액션 Trigger | 사용자 경험(UX) 목표 | 비주얼 변화/애니메이션 규칙 |
| :--- | :--- | :--- | :--- |
| **S0: 초기 (Empty)** | 페이지 로딩 / 첫 모듈 선택 전 | 관망적 상태. 다음 행동에 대한 기대감 조성. | 배경은 미세한 네온 시안(Cyan)의 그리드 패턴이 흐른다. 아크 영역은 비활성화되어 있다. |
| **S1: 조합 시작 (Active)** | 첫 번째 모듈 선택 완료 (`[모듈 A]`) | 자산 생성에 대한 몰입감 유도. | `[모듈 A]`가 중심 축으로 배치된다. 주변에 작은 빛의 파장이 퍼지기 시작한다. |
| **S2: 서사 연결 (Connecting)** | 두 번째 모듈 선택 완료 (`[모듈 B]`) | '연결'과 '발견'의 순간 강조. 가장 중요한 단계. | 1. `A`와 `B` 사이에 빛나는 아크(Arc)가 *시간차를 두고* 그려진다. (Stroke Animation 필수) 2. 아크 중앙에 **핵심 연결 키워드**가 짧게 플래시한다. |
| **S3: 깊이 증폭 (Deepened)** | 세 번째 모듈 이상 조합 완료 또는 최종 확정 시점 | '완성'과 '고유한 자산 소유권' 느낌 부여. | 1. 생성된 모든 아크가 하나의 거대한 빛의 구조물로 통합된다. 2. 배경 그리드 패턴이 사라지고, 전체 영역에 은은하고 안정적인 **네온 코어 광원**이 자리 잡는다. (최대 가치 상태) |

*   **타이포그래피:** 시스템 기본 산세리프 계열 (예: Pretendard, Inter). 간결하고 균일한 자간 유지.
*   **핵심 그래픽 에셋:** 순수하게 **선(Line)**과 **점(Dot)**으로 구성된다. 배경 패턴은 최소화한다.

| 요소 | 디자인 규칙 | 애니메이션 지시 (Dev Notes) |

### 🚀 핵심 코드 구현: `StoryArcCombiner.jsx`

이 컴포넌트는 사용자의 모든 상호작용을 중앙에서 관리하며, Designer가 정의한 State Machine (`S0 -> S1 -> S2 -> S3`)에 따라 렌더링 로직과 데이터 트래킹을 실행합니다.

```jsx
// src/components/StoryArcCombiner.jsx
import React, { useState, useCallback } from 'react';
import ModuleSelector from './ModuleSelector';
import ArcVisualizer from './ArcVisualizer';
import { useArcTracker } from '../hooks/useArcTracker';

// --- Mock API Functions (실제 환경에서는 백엔드 호출) ---
const trackEvent = (eventName, properties) => {
  console.log(`[TRACKING] Event: ${eventName}, Props:`, properties);
  // TODO: 실제 Mixpanel 또는 Amplitude 추적 코드 삽입
};

const simulateApiCall = async (data) => {
    await new Promise(resolve => setTimeout(resolve, 500)); // API 지연 시뮬레이션
    return { success: true, data: `Success for ${data}` };
};
// ---------------------------------------------------------


/**
 * StoryArcCombiner: 이름 모듈 조합의 상태 변화와 트래킹을 관리하는 메인 컴포넌트.
 */
const StoryArcCombiner = () => {
    const [modules, setModules] = useState([]); // 현재 선택된 모듈 배열
    const { updateState, currentDepth } = useArcTracker();

    // 1. 핵심 로직: 모듈 조합 및 상태 업데이트 핸들러
    const handleModuleSelect = useCallback(async (module) => {
        if (!module || modules.includes(module)) return;

        setModules(prev => [...prev, module]);
        
        // State Machine 실행 및 트래킹 전송
        updateState(module); 
        trackEvent('module_selected', { moduleName: module, currentDepth: modules.length + 1 });
    }, [updateState, modules.length]);


    // 2. Ownership Gap & Conversion Trigger (가장 중요한 부분)
    const handleFinalizeArc = async () => {
        if (modules.length < 2) {
            alert("서사를 완성하려면 최소 두 개의 모듈이 필요합니다.");
            return;
        }

        // S3 상태 도달 및 최종 트래킹 전송
        trackEvent('arc_finalization_attempt', { depth: modules.length, status: 'Attempted' });
        
        console.log("--- [Ownership Gap Detected] ---");
        alert(
            `축하합니다! ${modules.join(', ')}로 멋진 서사를 완성했습니다.\n\n` + 
            "이 서사 자산은 현재 *미완성* 상태입니다. 빛나는 아크를 영구적으로 소유하고, 모든 모듈의 잠재력을 열어보려면 유료 전용 서비스가 필요합니다."
        );

        // A/B 테스트 포인트: 여기서 결제 페이지(Payment Gateway)로 리다이렉트 로직 실행
        // 예시: history.push('/upgrade-story-arc'); 
    };


    return (
        <div className="storybook-container">
            <h1>✨ StoryArc Module Combiner</h1>

            {/* 1. 시각화 영역 */}
            <ArcVisualizer modules={modules} depth={currentDepth} />
            
            {/* 2. 모듈 선택 및 상호작용 영역 */}
            <div className="module-selector">
                <h2>모듈 조합 (State Change Input)</h2>
                <ModuleSelector onSelect={handleModuleSelect} />
            </div>

            {/* 3. 액션 버튼: 상태 S3 도달 지점이자, 구매 유도 포인트 */}
            <button 
                onClick={handleFinalizeArc} 
                disabled={modules.length === 0}
                className="cta-btn"
            >
                서사 아크 완성 및 소유권 증명 (🔥)
            </button>

        </div>
    );
};

export default StoryArcCombiner;
```

```jsx
/**
 * useArcTracker: Story Arc 조합 과정을 추적하는 커스텀 훅.
 * 모든 상태 변화(S0 -> S3)를 관리하고, 다음 단계를 결정합니다.
 */
export const useArcTracker = () => {
    // [모듈 A, 모듈 B, ...] 형태의 배열을 사용하여 현재 깊이와 순서를 추적
    const [history, setHistory] = useState([]); 

    // currentDepth: 현재 서사 자산의 깊이 (1부터 시작)
    const currentDepth = history.length;

    /**
     * stateUpdate: 모듈 선택 시 호출되며, State Machine 로직을 수행합니다.
     * @param {string} newModule - 새로 추가된 모듈 이름 또는 데이터 객체
     */
    const updateState = useCallback((newModule) => {
        setHistory(prevHistory => {
            const updatedHistory = [...prevHistory, newModule];

            let nextState = 'S0: 초기 (Empty)'; // 기본 상태
            let stateDescription = '';

            if (updatedHistory.length === 1) {
                nextState = 'S1: 조합 시작 (Active)';
                stateDescription = `[${newModule}] - 첫 번째 요소 배치 완료.`;
            } else if (updatedHistory.length === 2) {
                // S2 로직 발동: 두 모듈 사이의 아크 생성
                nextState = 'S2: 서사 연결 (Connecting)';
                stateDescription = `[${updatedHistory[0]}] $\\xrightarrow{빛나는 아크}$ [${newModule}] - 핵심 키워드 발견.`;
            } else { 
                // S3 로직 발동: 깊이 증폭
                nextState = 'S3: 깊이 증폭 (Deepened)';
                stateDescription = `모든 모듈 연결 완료! ✨ 서사적 자산 완성 단계 진입.`;
            }

            console.log(`[STATE CHANGE] ${nextState}: ${stateDescription}`);
            return updatedHistory;
        });
    }, []);


    return { updateState, currentDepth };
};
```

아래 코드는 서비스의 핵심 컴포넌트인 **모듈 시뮬레이터** 내부에 위치할 게이트(Gateway) 역할을 수행합니다.

```jsx
const useArcTracker = () => {

        isGapDetected: false, // Ownership Gap 감지 여부 (True일 때 유료 전환 필수)

    // 1. 상태 정의
    const [isLoading, setIsLoading] = useState(false);
    const [gapMessage, setGapMessage] = useState(''); // Ownership Gap 발생 시 노출될 메시지

    /**
     * Ownership Gap 감지 및 결제 플로우 시작 함수
     */
    const handleAttemptCompletion = useCallback(async () => {
        if (!isGapDetected || isLoading) return;

        setIsLoading(true);
        setGapMessage(`⚠️ 경고: 현재 자산의 서사적 깊이(${currentStoryDepth}%)는 완벽하지 않습니다. 전문적인 '완성'을 위해 유료 패키지를 고려해 보세요.`);
        
        try {
            // 3. API 호출 시뮬레이션 및 데이터 트래킹 (핵심)
            const result = await callPaidCompletionAPI('StoryAsset', currentStoryDepth);

            if (result.success) {
                setGapMessage(`✅ 성공! ${result.message} 결제 페이지로 이동합니다.`);
                // 4. 성공 시, 상위 컴포넌트의 콜백을 호출하여 리디렉션 유도
                onCompleteSuccess(result.token); 
            } else {
                setGapMessage(`❌ 실패: ${result.error} (다시 조합하거나 다른 모듈을 사용해 보세요.)`);
            }

        } catch (error) {
            console.error("Payment Flow Error:", error);
            setGapMessage("시스템 오류가 발생했습니다. 잠시 후 다시 시도해주세요.");
        } finally {
            setIsLoading(false);
        }
    }, [isGapDetected, currentStoryDepth]);

    // Story Depth가 특정 임계점(예: 40~75 사이)에 도달하고 아직 유료 전환을 시도하지 않았다면 게이트 활성화
    useEffect(() => {

            // 이 로직이 핵심: 사용자가 깊이를 쌓을수록 시스템이 '개입'하여 Gap을 노출시킴
            console.warn(`[Gate Activated] Story Depth ${currentStoryDepth}% reached the critical transition zone.`);
        }
    }, [currentStoryDepth, isGapDetected]);

    // 6. 렌더링 로직
    return (

                onClick={handleAttemptCompletion} 
                disabled={isLoading || !isGapDetected}
                style={styles.ctaButton}
            >

            </button>

                *(이 버튼을 누르는 순간, 당신의 데이터는 아카이브에 기록됩니다.)*
            </p>
        </div>
    );
};
```
